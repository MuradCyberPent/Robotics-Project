void loadCustomers()

void registerCustomer()
{
    char username[MAX_NAME_LEN];
    char email[MAX_EMAAIL_LEN];
    int isUnique;
    do
    {
        isUnique = 1;
        printf("Enter username: ");
        scanf("%s", username);
        for (int i = 0; i < totalCustomers; i++)
        {
            if (strcmp(customers[i].username, username) == 0)
            {
                isUnique = 0;

                printf("Error: Username already exists. Please choose a different username.\n");
                break;
            }
        }
    }while (!isUnique);
    clearScreen();

    do
    {
        isUnique = 1;
        printf("Enter email address: ");
        scanf("%s", email);


        for (int i = 0; i < totalCustomers; i++)
        {
            if (strcmp(customers[i].email, email) == 0)
            {
                isUnique = 0;
                printf("Error: Email address already exists. Please use a different email.\n");
                break;
            }
        }
    } while (!isUnique);
    clearScreen();

    strcpy(customers[totalCustomers].username, username);
    strcpy(customers[totalCustomers].email, email);

    printf("Enter password: ");
    scanf("%s", customers[totalCustomers].password);

    totalCustomers++;
    saveCustomers();
    clearScreen();
    printf("------------ Registration successful! -------------\n\n");
}

void addToCart()
{
    clearScreen();

        displayMedicines();
        return;
    }
    displayMedicines();
    int id, found = 0;
    printf("Enter Medicine ID to add to cart: ");
    scanf("%d", &id);
    clearScreen();
    for (int i = 0; i < totalMedicines; i++)
    {
        if (medicines[i].id == id)
        {
            if (medicines[i].quantity > 0)
            {
                cart[cartSize++] = medicines[i];
                saveMedicines();
                printf("Medicine added to cart!\n");
                for (int j = 0; j < cartSize; j++)
                {
                    printf("ID: %d, Name: %s, Price: %.2f\n", cart[j].id, cart[j].name, cart[j].price);
                }
            }
            else
            {
                printf("Medicine is out of stock!\n");
            }
            found = 1;
            break;
        }
    }
    if (!found)
    {
        printf("Medicine ID not found!\n");
    }
    printf("\n-------------------------\n");
}

void buyMedicine()
{
    clearScreen();
    float totalBill = 0;
    FILE *file = fopen(FILE_PATH "buy_details.txt", "a");
    if (file == 0)
    {
        printf("Error saving buy details.\n");
        return;
    }

    for (int i = 0; i < cartSize; i++)
    {
        for (int j = 0; j < totalMedicines; j++)
        {
            if (medicines[j].id == cart[i].id)
            {
                if (medicines[j].quantity > 0)
                {
                    medicines[j].quantity--;
                    totalBill += cart[i].price;
                }
                else
                {
                    printf("Not enough stock for %s!\n", cart[i].name);
                    fclose(file);
                    return;
                }
            }
        }
    }
    if(totalBill == 0.0){
        printf("Medicine has not been added!\n\n");
        printf("----------------------------\n");
        return 0;
    }
    fprintf(file, "Customer: %s, Total Bill: %.2f\n", customers[totalCustomers - (totalCustomers - cnt)].username, totalBill);
    for (int j = 0; j < cartSize; j++)
    {
        fprintf(file, "  ID: %d, Name: %s, Price: %.2f\n", cart[j].id, cart[j].name, cart[j].price);
    }
    fprintf(file, "  -----------------------------\n");
    fclose(file);
    saveMedicines();
    cartSize = 0;
    printf("Your total bill is %.2f\nThank you for your purchase!\n\n", totalBill);
    printf("-----------------------------\n");
}
